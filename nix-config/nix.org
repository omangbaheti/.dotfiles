#+name: makeSystem
#+begin_src nix
mkSystem = machine:
  let
    pkgs = nixpkgs.legacyPackages.${machine.system};
    isWSL = machine.systemType == "nix-wsl";
  in
    nixpkgs.lib.nixosSystem 
      {
        system = machine.system;
        modules = 
          [
            (if isWSL then nixos-wsl.nixosModules.wsl else {})
            home-manager.nixosModules.home-manager
            # Forward args into HM modules here:
            ({ ... }:
              {
                home-manager.extraSpecialArgs = {
                  inherit machine;
                  inherit inputs;
                  stable = stableFor machine.system;
                  host = machine.host;
                  username = machine.username;
                  systemType = machine.systemType;
                  machines = machines;
                };
                # Per-package pins live here. Empty lock = no-op until you add
                # something with `mvs lock add`; see "Updating Packages".
                home-manager.sharedModules = [
                  inputs.multiverse.homeManagerModules.default
                  {
                    multiverse = {
                      enable = true;
                      config.allowUnfree = true;
                      lock = ./multiverse.lock;
                    };
                  }
                ];
                home-manager.users.${machine.username} = import ./${machine.host}/home.nix;
              })
            lanzaboote.nixosModules.lanzaboote  
            # Same lock file, system side: `multiverse.locked.<attr>` is
            # available to configuration.nix and installs into
            # environment.systemPackages automatically.
            inputs.multiverse.nixosModules.default
            { multiverse = { enable = true; config.allowUnfree = machine.allowUnfree; lock = ./multiverse.lock; }; }
            ./${machine.host}/configuration.nix
            nixarr.nixosModules.default 
          ];

        specialArgs = {
          inherit inputs; 
          inherit self;
          machine = machine;
          stable = stableFor machine.system;
          host = machine.host;
          username =  machine.username;
          systemType = machine.systemType;
        };
      };
#+end_src

mkHome = machine:
  home-manager.lib.homeManagerConfiguration {
    pkgs = nixpkgs.legacyPackages.${machine.system};
    extraSpecialArgs = 
      {
        inherit inputs;
        machine = machine;
        stable = stableFor machine.system;
        host = machine.host;
        username = machine.username;
        systemType = machine.systemType;
      };

    modules = [
      ({ ... }: {
        home.username = machine.username;
        home.homeDirectory = "/home/${machine.username}";
        # home.homeDirectory = if machine.username == "root" then "/root" else "/home/${machine.username}"; 
        # home-manager.useGlobalPkgs = true;
        # home-manager.useUserPackages = true;
      })

      inputs.multiverse.homeManagerModules.default
      {
        multiverse = {
          enable = true;
          config.allowUnfree = true;
          lock = ./multiverse.lock;
        };
      }

      # Put your actual HM module(s) here
      ./${machine.host}/home.nix
    ];
  };

* Updating Packages

Two update paths, deliberately separate. The base moves rarely and in small
steps; individual packages move on their own, whenever you want them to.

| what                        | command                        | what it downloads          |
|-----------------------------+--------------------------------+----------------------------|
| everything (weekly)         | ~nix flake update nixpkgs~     | only what 26.05 backported |
| one package                 | ~mvs lock update <attr>~       | that package's closure     |
| every pin, newest available | ~mvs lock update --all~        | every pin's closure        |
| nothing, just look          | ~mvs lock status~              | nothing                    |

** One-time: get the ~mvs~ CLI

Nothing is installed by the flake; run it out of the multiverse repo, from the
directory holding ~multiverse.lock~:

#+begin_src console
nix run 'github:fzakaria/nixpkgs-multiverse#mvs' -- lock status
#+end_src

** Adding a package to the per-package list

#+begin_src console
# newest version the index knows
nix run 'github:fzakaria/nixpkgs-multiverse#mvs' -- lock add obsidian

# or pin a specific version (prefix match: 1.10 accepts 1.10.x)
nix run 'github:fzakaria/nixpkgs-multiverse#mvs' -- lock add libreoffice@25.8

nixos-rebuild switch --flake .#fern
#+end_src

The result is ~config.multiverse.locked.<attr>~ and ~config.multiverse.pinned.<attr>~,
for the options that take a package rather than installing one:

#+begin_src nix
programs.obsidian.package = config.multiverse.locked.obsidian;
#+end_src

A lock file set in the module is installed automatically, so ~home.packages~
does not need to list it again.

** Updating one package, and only that package

#+begin_src console
nix run 'github:fzakaria/nixpkgs-multiverse#mvs' -- lock status
# ATTR       PINNED   LATEST   BEHIND
# obsidian   1.9.10   1.10.2   3 versions, 41 days

nix flake update multiverse        # learn about newer revisions first
nix run 'github:fzakaria/nixpkgs-multiverse#mvs' -- lock update obsidian
nixos-rebuild switch --flake .#fern
#+end_src

~lock update <attr>~ rewrites exactly one entry in ~multiverse.lock~; every other
pin keeps its revision. This is the thing a single flake input cannot do.

** Keeping the pins cheap

Each distinct revision in the lock is one nixpkgs fetch and evaluation.
~lock minimize~ moves the pins onto the fewest revisions that still serve the
versions already pinned — versions never change, only which build of them:

#+begin_src console
nix run 'github:fzakaria/nixpkgs-multiverse#mvs' -- lock minimize
#+end_src

** Two lock files: global and gui

A pin is not scoped to a host — the module installs every entry it is given,
so *the file you point the module at is the scope*. Two files, two audiences:

| lock file              | read by                          | reaches                    |
|------------------------+----------------------------------+----------------------------|
| ~multiverse.lock~      | the module, in every host        | all machines               |
| ~multiverse-gui.lock~  | ~modules/gui-packages.nix~ only  | fern and nyx               |

~multiverse.lock~ is auto-installed: adding obsidian there puts it on every
host, system and user, with no ~home.packages~ line anywhere.

~multiverse-gui.lock~ is read by hand inside ~gui-packages.nix~, which only
fern and nyx import, so its contents stay on those two:

#+begin_src console
cd ~/.dotfiles/nix-config
git pull
nix run 'github:fzakaria/nixpkgs-multiverse#mvs' -- --file multiverse-gui.lock lock add libreoffice@26.2
#+end_src

Then delete ~libreoffice~ from the ~home.packages~ list in the same file —
~guiPins~ appends every entry of that lock already.

Once a year, or when you notice something stale, both files need attention;
~lock status~ takes ~--file~ too:

#+begin_src console
nix run 'github:fzakaria/nixpkgs-multiverse#mvs' -- lock status
nix run 'github:fzakaria/nixpkgs-multiverse#mvs' -- --file multiverse-gui.lock lock status
#+end_src

Rule of thumb: anything in ~common-packages.nix~ or shared GUI code belongs in
the global lock; anything device-specific belongs in that device's ~home.nix~
on the base channel, or in the gui lock if the whole shared file wants it.

** Limits worth knowing before you pin something

- A pin comes from a *different* nixpkgs than the running system, so it brings
  that revision's closure, including its own libc. Right for leaf applications
  and CLI tools; wrong for libraries other packages link against.
- Only top-level attributes are indexed (~jetbrains.idea~ and a few other
  nested sets work). ~python3Packages.*~ and ~emacsPackages.*~ do *not*, so
  ~emacs-pgtk~ is pinnable but the elpa packages built from the base nixpkgs
  are not — leave emacs on the base.
- Stored paths still substitute from cache.nixos.org, so a pin is a download,
  not a compile.
- aarch64 machines get their own copies; a pin downloads there separately.

# PATH=/nix/store/5qng39wihv3lfgr03cf7mqbg4lpf4m45-cmake-3.30.5/bin:/mnt/c/Windows/System32/WindowsPowerShell/v1.0:$PATH
function isWinDir 
{
    case "$PWD/" in
        /mnt/*) return 0 ;;
        ,*) return 1 ;;
    esac
}

function lazygit 
{
    if isWinDir; then
	#  Use Windows `lazygit.exe` in Windows-mounted dirs
        command lazygit.exe "$@"
    else
	#  Use native Linux `lazygit` in native dirs
        command lazygit "$@"
    fi 
}
